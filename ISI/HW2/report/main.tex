\section{سوال ۳}

در هرمورد باید بررسی شود با میل کردن 
\lr{n}
به سمت بینهایت، مقدار 
$\cfrac{1}{n}S_n$
به سمت 
$1 - p$
میل می‌کند یا خیر.
در حالت کلی می‌توان گفت:
\begin{latin}
	$
	S_n = \sum x_i (1 - p_i)
	$
\end{latin}
که در آن $p_i$ ها از توزیع برنولی با متغیر $p$ هستند. در هرمورد باید چک کرد تعداد متغیرها به سمت بینهایت میل می‌کند یا خیر و همچنین واریانس $S_n$ مقدار متناهی باشد.
\begin{enumerate}



 	\item 
 	در اینجا عملا هدف میانگین گرفتن \lr{n} متغیر تصادفی از توزیع برنولی است پس:
 	\begin{latin}
 		$
 		\cfrac{1}{n}	E(S_n) = \cfrac{1}{n} \sum (1 - p) = \cfrac{n(1 - p)}{n} = 1 - p
 		$
 	\end{latin}
 	واریانس نیز یک کسر با مخرج 
 	$\sqrt{n}$
 	و صورت ثابت 
 	($p(1 - p)$)
 	می‌باشد پس با زیاد شدن \lr{n} به صفر میل می‌کند. پس قانون اعداد بزرگ در اینجا صدق می‌کند.
 	\item 
 	در اینجا نیز با زیاد شدن 
 	\lr{n}
 	تعداد متغیرهای تصادفی زیاد می‌شوند. همچنین واریانس و امید مشابه قسمت قبل حساب می‌شوند با این تفاوت که تعداد متغیرهای تصادفی برابر 
 	$
 	\cfrac{n}{2}
 	$
 	است.
 	
\begin{latin}
	$
	\cfrac{1}{n}	E(S_n) = \frac{1}{n} \sum\limits_{i = 1}^{\frac{n}{2}} 2(1 - p)  = 1 - p
	$
\end{latin}
واریانس اینجا نیز مشابه قسمت قبل است (می‌توان فرض کرد که این توزیع درواقع حاصل دوبار جمع کردن توزیع قبل است و چون 
\lr{n}
به سمت بینهایت می‌رود مسئله‌ای پیش نمی‌آید و دوبرابر صفر، صفر است.) و برابر صفر می‌شود. پس قانون اعداد بزرگ دراینجا صدق می‌کند.
\item 
در اینجا با زیاد شدن \lr{n} تعداد متغیرهای تصادفی تغییری نمی‌کند، پس قانون اعداد بزرگ در اینجا صدق نمی‌کند زیرا عملا تعداد متغیرهای تصادفی ثابت و برابر دو می‌باشد. حتی واریانس نیز به سمت صفر میل نمی‌کند. 
\begin{latin}
	$
	\mu = E(x) = (1 - p)^2 n + 2p(1 - p)(\cfrac{n}{2})  + 0 * p^2= n(1 - p)(1 - p + p) = n(1 - p)
	$
	\\
	$
	Var(S_n) = E((x - \mu)^2) = (1 - p)^2 (n - \mu)^2 + 2p(1 - p)(\cfrac{n}{2} - \mu)^2 + p^2 \mu^2 = \cfrac{n^2}{2}p(1 - p)
	$
	\\
	$
	\text{mean}: Var(\cfrac{S_n}{n}) = \cfrac{Var(S_n)}{n^2} = p(1 - p) 
	$
\end{latin} 
که یعنی با زیاد شدن \lr{n} واریانس میانگین به صفر میل نمی‌کند پس قانون اعداد بزرگ برقرار نیست.  (به غیر از حالات خاص $p = 0$ و $p = 1$)

\item 
اینجا نیز مشابه قسمت قبل چون تعداد متغیرهای تصادفی ثابت و برابر یک است قانون اعداد بزرگ صدق نمی‌کند. واریانس نیز برابر است با:
\begin{latin}
	$
	\mu = E(X) = \sum xf(x) = n(1 - p)
	$
	\\
	$
	Var(S_n) = E((X - \mu)^2) = (1 - p) (n - n(1 - p))^2 = (1 - p)(pn)^2 = n^2 * p^2 * (1 - p)
	$
	\\
	$
	\text{mean}: Var(\cfrac{S_n}{n}) = \cfrac{Var(S_n)}{n^2} = p^2 ( 1 - p)
	$
\end{latin}
که در اینجا نیز واریانس به سمت صفر میل نمی‌کند و قانون صدق نمی‌کند (به غیر از حالات خاص $p = 0$ و $p = 1$)
\end{enumerate}≈

\subsubsection{۱}
\begin{latin}

\end{latin}
برای محاسبه 
\lr{MGF}
متغیر 
\lr{X}
چون می‌دانیم مشتق \lr{i} ام تابع 
\lr{MGF}
برابر ممان \lr{i} ام است داریم:
\begin{latin}
	~\\
	$
	 M_X(t) \approx M_X(0) + tM_X'(0) + t^2 M_X''(0) 
		= M_X(0) + tE(X)  + \dfrac{t^2}{2} E(X^2) 
	$
	\\
	$
	\xrightarrow{E(X) = 0, E(X^2) = 1}  M_X(t) \approx  M_X(0) + 0 + \dfrac{t^2}{2} 
	\xrightarrow{M_X(0) = E(e^0) = E(1) = 1} M_X(t) \approx 1  + \dfrac{t^2}{2} 
	$
\end{latin}

\subsubsection{۲}
داریم:
\begin{latin}
	$
	M_{Z_n}(t) = M_{\sqrt{n}\bar{X_n}} = E(e^{t\sqrt{n}\bar{X}}) = E(exp(t\sqrt{n}\bar{X})) = E(exp(t\sqrt{n} (\dfrac{\sum X_i}{n})))
	= E(\prod  exp(t\sqrt{n} (\dfrac{ X_i}{n})))  = E(\prod  exp(t(\dfrac{ X_i}{\sqrt{n}}))) 
	= E(\prod\limits_{i = 1}^n e^{\dfrac{tX_i}{\sqrt{n}}})
	$
\end{latin}
چون متغیرها 
مستقل
هستند می‌توان امید ضرب‌ها را به ضرب امید تبدیل کرد و چون  یکسان هستند، می‌توان فقط برای یک متغیر حساب کرد و سپس آن را بتوان 
\lr{n}

\begin{latin}
	$
	M_{Z_n}(t) = E(exp(\dfrac{tx}{\sqrt{n}}))^n = M_X(\dfrac{t}{\sqrt{n}})^n
	$
\end{latin}	
که این همان چیزی است که میخواستیم.

\begin{latin}

	lim_{n \rightarrow \infty} M_{Z_n}(t) = lim_{n \rightarrow \infty} (1 + \dfrac{(\dfrac{t}{\sqrt{n}})^2}{2} )^n = (1 + \dfrac{t^2}{2n})^n 
	$

	$
\end{latin}

\subsubsection{۲}
از آن جا که عملگرهای خطی روی انتگرال (و بالطبع امید) تاثیر ندارند می‌توان عبارت را در مخرج ضرب کرد و با میانگین جمع کرد. پس باتوجه به فرمول‌های میانگین و واریانس توزیع داریم:
\begin{latin}
	$
	\cfrac{\bar{X} - \mu}{\cfrac{\sigma}{\sqrt{n}}} \sim N(0, 1) \Rightarrow \bar{X} \sim \cfrac{\sigma}{\sqrt{n}} N(0, 1) + \mu \Rightarrow \bar{X} \sim N(\mu, \cfrac{\sigma}{\sqrt{n}}) 
	$
\end{latin}
با توجه به قضیه حد مرکزی، توزیع

\bar{X_n}
$
از نوع نرمال می‌باشد.

\lr{CLT}
درباره توزیع نمونه‌برداری صحبت می‌کند. توزیع میانگین نمونه‌ها. با این‌که خود میانگین نمونه باتوجه به توزیع بدست می‌آید (برای مثال در بخش‌های قبل به ازای $n=2$ حساب شد.) اما توزیع این میانگین‌ها به شکل نرمال خواهد بود.
