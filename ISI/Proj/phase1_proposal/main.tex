	\lr{\section{Data Provenance and Structure}}
	مجموعه داده انتخاب شده
	\cite{lending_club_kaggle}
	، شامل درخواست افراد برای وام است و 
	\lr{label}
	آن این است که با درخواست وام موافقت شده است یا خیر.  
	
	در نسخه‌ای که دانلود شده است، داده بین سال‌های ۲۰۰۷ تا ۲۰۱۸ جمع‌آوری شده است. این داده توسط خود سایت 
	\lr{Lending Club}
	از تهیه شده است و سپس در 
	\lr{Kaggle}
	توسط یکی از کاربران سرهم شده است.
	
	داده به اینصورت جمع‌آوری شده است که ابتدا 
	\lr{Lending Club}
	داده مشتریان خود را جمع‌آوری کرده است و سپس با استفاده از داده‌های دیگیری مربوط به اعتبار افراد، مجموعه داده غنی‌تر شده است.
	\lr{\section{Suitability for Statistical Inference}}
	این مجموعه داده بزرگ است و حدود ۲۹۹۰۹۴۴۲ سطر دارد. ستون‌های آن انواع مختلفی دارند.  نمایی از این تنوع در جدول  
	\ref{tab:lendingclub_features}
	آمده است. 
	\subsection{نمونه‌برداری و توزیع نمونه }
	به دلیل حجم زیاد این مجموعه داده و این موضوع که داده‌های این مجموعه داده تعریف مشخصی دارند و حد و حدود آن‌ها مشخص است، احتمالا فرض‌هایی مثل محدود بودن واریانس در آن صدق کند و قضایایی مثل قضیه اعداد بزرگ و حد مرکزی در آن صدق کند. 
	


	
	\subsection{ویژگی‌ها}
		تعداد ستون‌های این مجموعه داده نیز زیاد است و یحتمل فرض‌هایی مثل مستقل خطی بودن در آن صدق نکند. اگر برای ستون‌های عددی آن ماتریس 
	\lr{Correlation}
	حساب شود و سپس توزیع مقادیر 
	\lr{Correlation}
	را ببینیم می‌توان دید مقادیر قدر مطلق  
	\lr{Correlation}
	تقریبا همه بازه بین صفر تا یک را پوشش می‌دهند و هم ویژگی‌هایی داریم که به هم وابسته هستند و هم ویژگی‌هایی که هیچ ارتباطی به یکدیگر ندارند.
	

	
	\subsection{توزیع ویژگی‌ها}
	اکثر ویژگی‌های عددی، توزیعی شبیه به توزیع نرمال دارند ولی چولگی به سمت راست دارند که باید این موضوع در آن‌ها برطرف شود. همچنین بازه ویژگی‌ها یکی نیست که می‌تواند مشکل‌ساز شود و بهتر است که اول توزیع‌ها را نرمال استاندارد کرد.
	
	\lr{\section{Scope of Analysis}}
	چون این مجموعه داده سطرهای زیادی دارد و هم ویژگی‌های عددی پیوسته و هم ویژگی‌های 
	\lr{categorial}
	دارد،
	می‌توان کارهای متعددی روی آن انجام داد.
\lr{\subsection{Hypothesis Testing}}
	می‌توان چک کرد که 
		(\lr{\texttt{verification\_status}})
	بر روی 
	(\lr{\texttt{int\_rate}})
	تاثیر دارد یا خیر. فرض صفر این است که تاثیری ندارد. با آزمون فرضیه اگر فرض صفر رد شود، یعنی احتمالا  وضعیتی که از کاربر بدست می‌آید روی نرخ سود او تاثیر می‌گذارد، ولی اگر فرض صفر رد نشد، نمی‌توان حرفی درباره رابطه بین وضعیت تایید کاربر و نرخ سود او گفت.
	
\lr{\subsection{ANOVA}}
با استفاده از 
\lr{ANOVA}
می‌توان مقایسه کرد که حالت‌های متفاوت 
	(\lr{\texttt{home\_ownership}})
	چه تاثیری روی میانگین
	(\lr{\texttt{Debt\_to\_Income}})
	دارد. زیرا مالکیت خانه، یک ویژگی رده‌ای
	\LTRfootnote{Categorical}
	است و با استفاده از 
	\lr{ANOVA}
	می‌توان دید که روی 
	\lr{Debt to Income}
	تاثیر دارد یا خیر.

\lr{\subsection{Chi Squared}}
برای این قسمت باید از دو داده که رده‌ای هستند استفاده کرد. برای مثال می‌توان رابطه 
	(\lr{\texttt{home\_ownership}})
	با
		(\lr{\texttt{grade}})
		را بررسی کرد.
\lr{\subsection{Linear Regression}}
برای 
\lr{linear regression}
می‌توان سعی کرد که میزان درآمد سالانه 
	(\lr{\texttt{annual\_income}})
	را پیش‌بینی کرد. میزان درآمد یک متغیر پیوسته است و رگرشن برای این کار مناسب است. یکی از بخش‌های این قسمت، انتخاب ویژگی‌های مناسب برای رگرشن است. 

\lr{\subsection{Non parametric test}}
در اینجا می‌توان رابطه بین 
	(\lr{\texttt{loan\_amnt}})
	و
		(\lr{\texttt{loan\_status}})
		را بررسی کرد که آیا مبلغ وام در نرخ نکول آن تاثیر دارد یا خیر. چون مبلغ وام چولگی دارد، اگر به شکل نرمال درنیاید، تست‌های پارامتریک مثل
	\lr{T Test}
	برای آن مناسب نیستند و تست  
	\lr{non parametric}
	بهتر است.
