	\section*{بخش دوم: تحلیل داده‌های رفتاری}
	
	\subsection*{ساختار داده‌ها}
	داده‌های داده شده شامل
	داده‌های ۴۰ آزمودنی (۲۰ مرد و ۲۰ زن) در پوشه‌های جداگانه است. 	تنها تفاوت قابل مشاهده‌ای که در این داده‌ها دیده شد این است که بلوک سوم بجای ۴۵ دور، شامل ۶۰ دور می‌باشد.  برای بارگذاری داده، همه داده‌ها در یک 
	\lr{Dataframe}
	قرار گرفتند که یک ستون جنسیت نیز دارد. 
	پس از بارگذاری و پردازش، ستون‌های زیر به داده‌ها اضافه شدند:
	\begin{itemize}
		\item \textbf{\lr{block}}: شماره بلوک بر اساس شماره‌ی \lr{trial}
		\item \textbf{\lr{trial\_in\_block}}: شماره دور در هر بلوک (شروع از ۱)
		\item \textbf{\lr{chosen\_probability}}: احتمال پاداش کارت انتخاب شده
		\item \textbf{\lr{optimal\_choice}}: آیا انتخاب بهینه بوده است (۱/۰)
		\item \textbf{\lr{rewarded}}: آیا پاداش دریافت شده است (۱/۰)
	\end{itemize}

	\section*{بخش سوم: مدل‌سازی محاسباتی}
	
	\subsection*{معرفی مدل‌ها}
	مدل‌های \lr{Q-Learning} با سه تابع سیاست مختلف پیاده‌سازی شدند.
	
	\subsubsection*{تابع ارزش (\lr{Q-Value})}
	\begin{equation}
		Q_{t+1}(a) = Q_t(a) + \alpha \cdot (r_t - Q_t(a))
	\end{equation}
	که در آن:
	\begin{itemize}
		\item $Q_t(a)$: ارزش تخمینی برای عمل $a$ در زمان $t$
		\item $\alpha$: نرخ یادگیری (\lr{Learning Rate})
		\item $r_t$: پاداش دریافت‌شده
	\end{itemize}
	
	\subsubsection*{سیاست‌های انتخاب عمل}
	
	\textbf{۱. \lr{Softmax}:}
	\begin{equation}
		P(a) = \frac{\exp(\beta \cdot Q(a))}{\sum_{i} \exp(\beta \cdot Q(i))}
	\end{equation}
	که $\beta$ پارامتر دما (\lr{Temperature}) است. هرچقدر این مقدار بیشتر باشد مدل کمتر \lr{explore} می‌کند و بیشتر حریصانه عمل می‌کند. 
	
	\textbf{۲. \lr{Epsilon-greedy}:}
	با احتمال $\epsilon$ یک عمل تصادفی و با احتمال $1-\epsilon$ بهترین عمل انتخاب می‌شود.
	
	\textbf{۳. \lr{UCB} (\lr{Upper Confidence Bound}):}
	\begin{equation}
		UCB(a) = Q(a) + \beta \cdot \sqrt{\frac{\ln(N)}{n_a}}
	\end{equation}
	که در آن $N$ تعداد کل انتخاب‌ها و $n_a$ تعداد دفعات انتخاب عمل $a$ است.
	یعنی این مدل علاوه بر اینکه سعی می‌کند تا بهترین کار را انجام بدهد،‌به عمل‌هایی که کمتر انجام شده وزن بیشتری می‌دهد تا مدل هم 
	\lr{explore}
	کند و هم 
	\lr{exploit}
	
	
	
	\subsection*{روش برازش مدل‌ها}
	برای هر شرکت‌کننده و هر بلوک، سه مدل با توابع سیاست مختلف برازش شدند. فرایند برازش شامل موارد زیر بود:
	\begin{enumerate}
		\item بهینه‌سازی پارامترها با روش \lr{L-BFGS-B}
		\item استفاده از ۲۰ شروع تصادفی برای یافتن بهینه‌ی سراسری
		\item محاسبه‌ی معیارهای \lr{AIC} و \lr{BIC} برای مقایسه‌ی مدل‌ها
	\end{enumerate}

	\subsubsection*{توزیع پارامترها}
	توزیع پارامترها در شکل 
	\ref{fig:parameter_dist}
	آمده است. رفتار عجیب مدل 
	\lr{UCB}
	را می‌توان مشاهده کرد که برخلاف انتظار که یک مدل شبیه نرمال را انتظار داریم عمدتا توزیع‌هایی است که در میانگین چگالی کمتری وجود دارد. این موضوع برای نرخ یادگیری 
	\lr{softmax}
	هم وجود دارد. در مقابل، مدل 
	\lr{egreedy}
	رفتار نرمال‌تری از خود نشان می‌دهد. 
	
	\begin{itemize}
		\item 
		\textbf{$\alpha$}
		می‌توان دید که 
		$\alpha$
		در 
		\lr{softmax}
		توزیع نسبتا یکنواختی دارد و تنها در میانگین کمی چگالی کمتری دارد. این موضوع می‌تواند نشان دهد که در هر وضعیت مدل یا نمی‌خواسته چیزی یاد بگیرد و یا نیاز به یادگیری شدید بوده. این موضوع در سیاست
		\lr{UCB}
		نیز با شدت بیشتر مشاهده می‌شود. که توزیع بجای یک توزیع نرمال یک توزیع دو مدی است. این نشان می‌دهد که هر شرکت کننده چقدر از کار خود اثر می‌برد. اگر فرد تنها از وضعیتی که می‌بیند تصمیم خود را عوض کند، به 
		$\alpha$
		بالا می‌رسیم ولی اگر فارغ از نتیجه عمل کند 
		$\alpha$
		پایین خواهد بود. 
		
		در مقابل مدل 
		\lr{egreedy}
		نسبتا توزیع نرمالی دارد. این موضوع می‌تواند به این خاطر باشد که در سیاست 
		\lr{egreedy}
		مقدار 
		$\epsilon$
		می‌تواند روی 
		$\alpha$
		اثر بگذارد و برای همین شکل این توزیع فرق کرده است. 
		\item 
		\textbf{$\beta$}
		می‌توان دید که برای پارامتر $\beta$ سیاست 
		\lr{softmax}

		\lr{UCB}
		مجدد می‌بینیم که یک توزیع دومدی داریم. این موضوع در اینجا به نظر من به این علت است که سیاست 
		\lr{UCB}
		در این سوال نمی‌تواند به خوبی رفتار افراد را تقلید کند و از همین رو بهینه‌ساز مجبور به کاوش حالات حدی پارامتر شده است.
	\end{itemize}

	\subsubsection*{معیارهای بررسی}
	معیارهای 
	\lr{nll, aic}
	و
	\lr{bic}
	مورد بررسی قرار گرفتند. همراه با تعداد 
	\lr{mismatch}
	ها.
	\begin{itemize}
		\item 
		\textbf{\lr{nll}:}
		این معیار همان 
		\lr{negative log likelihood}
		است که نشان می‌دهد که چقدر احتمالات باهم تفاوت داشتند.
		
		\item 
		\textbf{\lr{AIC}:}
این معیار علاوه بر بررسی برازش مدل، پیچیدگی مدل را جریمه می‌کند. زیرا ممکن است یک مدل به علت پیچیدگی بیشتر بتواند احتمال را بهتر تخمین بزند ولی لزوما الگوی داده را یاد نگرفته باشد. برای همین علاوه بر 
\lr{nll}
مقدار را با منفی تعداد پارامترها جمع می‌کند. 



	\begin{equation}
		AIC = 2k - 2\ln(\hat{L})
	\end{equation}
	که $k$ تعداد پارامترها و $\hat{L}$ بیشینه‌ی تابع درستنمایی است.
	
	\item 
	\textbf{\lr{BIC}:}
	این مورد مشابه 
	\lr{AIC}
	است. با این تفاوت که از انجایی که سایز مدل ثابت است ولی 
	\lr{nll}
	با تعداد داده زیاد می‌شود، در داده زیاد 
	\lr{AIC}
	به همان 
	\lr{nll}
	متمایل می‌شود. پس برای حل این مشکل تعداد پارامترها را در لگاریتم تعداد داده ضرب می‌کند. 
	\begin{equation}
		BIC = k\ln(n) - 2\ln(\hat{L})
	\end{equation}
	که $n$ تعداد نقاط داده است.
		\end{itemize}
	
	در این مسئله خاص این معیارها خیلی باهم تفاوت ندارند. زیرا پیچیدگی مدل‌ها تقریبا برابر است و حجم داده نیز کم است. برای همین در نمودارها عمدتا فقط مقیاس فرق می‌کند و ترتیب و نسبت فاصله بین مدل‌ها برابر است. 
	
	مدل \lr{Softmax} در هر معیاری بهترین مدل است. این نشان می‌دهد که شرکت‌کنندگان از یک سیاست انتخاب نرم‌افزاری (\lr{Probabilistic}) پیروی می‌کنند تا یک سیاست قطعی.
	

	\section*{نتیجه‌گیری}
	با استفاده از مدل‌های محاسباتی می‌شد فهمید که شرکت کنندگان بیشتر از یک روش احتمالاتی پیروی کرده‌اند تا یک روش قطعی. باتوجه به نمودار 
	\ref{fig:entropy}
	می‌شد دید که روند به اینصورت نبوده که 
	\lr{explore}
	و
	\lr{exploit}
	جدا صورت گرفته باشد (مثلا اول همه کارت‌ها را امتحان کنند و بعد از شناسایی کارت بهینه تنها آن را انتخاب کنند.) بلکه همواره شرکت کنندگان بین انتخاب‌های مختلف سوییچ کرده‌اند. البته با پیروی از مواردی مثل 
	\lr{win stay lose shift}
	. همچنین این سوییچ کردن در جهت بهتر کردن میانگین کلی بوده است و بصورت تصادفی نبوده است.
