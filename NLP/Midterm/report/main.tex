	\subsection*{سوال ۱}
	\subsubsection*{راه‌حل}
	برای یک 
	\lr{tokenization} 
	با تعداد توکن کم می‌توان به 
	\lr{character tokenization}
	و برای یک 
	\lr{tokenization}
	با تعداد توکن زیاد می‌توان به 
	\lr{word tokenization}
	اشاره کرد. اگر تعداد 
	\lr{type}
	های توکن کم باشد، منجر به این می‌شود که احتمالاً توکن‌های پایه‌تری موجود است و توکن ناشناس پیدا نمی‌شود؛ ولی باعث می‌شود که خروجی فرآیند 
	\lr{tokenization}
	تعداد زیادی توکن باشد که فرآیند آموزش و استنتاج مدل را کند می‌کند. از طرف دیگر اگر تعداد 
	\lr{type}
	ها زیاد باشد، احتمال دیدن توکن ناشناس بیشتر است؛ برای مثال اگر مدل کلمات 
	\lr{drive, driver, teach}
	را دیده باشد ولی کلمه 
	\lr{teacher}
	را ندیده باشد، نمی‌تواند توکن مناسب برای کلمه 
	\lr{teacher}
	را بیابد.

	\subsection*{سوال ۳}
	\subsubsection*{راه‌حل}
	در طبقه‌بند
	\lr{Bag of Words Naive Bayes}
	دو فرض اساسی موجود است که اگر این دو فرض درست باشند، این طبقه‌بند همان طبقه‌بند بهینه 
	\lr{Bayes}
	می‌شود. 
	
	فرض اول در 
	\lr{Bag of Words}
	است که بیان می‌کند ترتیب کلمات اثری روی کلاس آن ندارد. 
	
	فرض دوم در قسمت 
	\lr{Naive Bayes}
	است که بیان می‌کند پدیدار شدن یک کلمه روی پدیدار شدن سایر کلمات اثری نخواهد داشت. 
	
	در عمل این دو فرض تقریباً هیچ‌وقت محقق نمی‌شوند. به عنوان شهود می‌توان دید که استفاده از 
	\lr{bigram}
	در مدل‌های \lr{N-gram} بهبود خیلی زیادی نسبت به مدل 
	\lr{unigram}
	دارد؛ پس فرض استقلال کلمات فرض دقیقی نیست.

	\subsection*{سوال ۵}
	\subsubsection*{راه‌حل}
	تسک پردازش احساسات در ظاهر یک طبقه‌بندی متن معمولی است، اما برخی چالش‌ها در آن برجسته‌تر هستند: 
	\begin{itemize}
		\item \textbf{کنایه:} یکی از چالش‌ها، کنایه و ایهام است. در تحلیل احساسات (مثلاً در کامنت‌های فیلم)، افراد ممکن است با تضاد نظر خود را بیان کنند (مثلاً: «این هم یکی از آن فیلم‌های شاهکاری بود که بعدش می‌خواهید پولتان را پس بگیرید!»).
		\item \textbf{ابهام تسک:} در بسیاری از موارد خودِ نظر فرد متناقض است و واضح نیست که برآیند نهایی مثبت است یا منفی.
		\item \textbf{زمینه (\lr{Context}):} در نظرات احساسی گاهی زمینه‌های بلندتری موجود است که استفاده از روش‌هایی با پنجره کوتاه مثل 
		\lr{5-gram}
		را بی‌اثر می‌سازد. 
	\end{itemize}
	\subsubsection*{علت}
	به نظرم مهم‌ترین بخش این اسلاید، راه‌های ارائه شده برای تسک طبقه‌بندی احساسات نیستند (چرا که احتمالا یک مدل زبانی مبتنی بر هوش مصنوعی از راه‌های ارائه شده دقت بالاتری بدهد.) بلکه توضیح تسک و نمونه جملاتی هستند که  در اسلاید آمده است. برای بررسی تسک، بهترین راه نیز بررسی ورودی و خروجی است که برای مثال 
	\lr{Negation}
	چه چالش جدیدی بیان می‌کند؟ و مهم دسته‌بندی این چالش‌ها است (نسبته به حفظ راه ارائه شده)
	\subsection*{سوال ۶}
	\subsubsection*{راه‌حل}
	در طبقه‌بند 
	\lr{Bernoulli}
	عملاً دو مقدار 
	$p$
	و
	$1 - p$
	برای حضور یا عدم حضور هر متغیر بررسی می‌شود. ولی در 
	\lr{Multinomial}
	تعداد تکرار متغیرها اهمیت دارد (در حالت \lr{Binarized} تکرار با عدد یک سقف‌گذاری می‌شود). این تفاوت بیشتر مفهومی است و در عمل نتایج طبقه‌بندی بسیار نزدیک است.
	\subsubsection*{علت}
	این عبارت که این دو طبقه‌بند یکی نیستند و باهم تفاوت دارند برای من سوال شد و جستجو کردم تا به جواب برسم. این که علیرغم شباهت ریاضیاتی، مفهوم این دو توزیع متفاوت است به نظرم نکته مهمی آمد. که حتی در 
	\lr{Binary Multivariate normal}
	علیرغم 
	\lr{clip}
	کردن، همچنان دید ما به متغیر مفهوم «تکرار» است نه «شکست» و یا «موفقیت».
	\subsection*{سوال ۷}
	\subsubsection*{راه‌حل}
	در مدل‌های 
	\lr{Generative}
	مدل به دنبال یادگیری توزیع احتمال مشترک $P(x, y)$ است و موقع استنتاج 
	\lr{likelihood}
	را حساب می‌کند. مانند 
	\lr{Naive Bayes}
	که توزیع هر کلاس را تخمین زده و با 
	\lr{MLE}
	پیش‌بینی می‌کند.
	در مدل‌های 
	\lr{Discriminative}
	مدل مستقیماً به دنبال یادگیری مرزهای تصمیم یا $P(y|x)$ است. مانند 
	\lr{Logistic Regression}
	که پارامترهای خط جداکننده را یاد می‌گیرد.
	\subsubsection*{علت}
	این سوال درواقع ذات مدل‌ها را بررسی می‌کند. که مزیت و عیب هر کدام از روش‌ها چیست. برای من برای اولین بار در ذهنم این سوگیری وجود داشت که مدل‌های قوی 
	توزیع را یاد می‌گیرند و 
	\lr{generative}
	هستند و مدل‌های ضعیف‌تر 
	\lr{discriminative}
	. ولی می‌بینیم که یک مدل خیلی ساده مثل 
	\lr{naive bayes}
	نیز می‌تواند \lr{generative } باشد و اصل تفاوت در رویکرد است نه پیچیدگی.
	\subsection*{سوال ۸}
	\subsubsection*{راه‌حل}
	چون در 
	\lr{Logistic Regression}
	خروجی حاصل یک ترکیب خطی از ویژگی‌هاست، وزن اختصاص داده شده به هر ویژگی نشان‌دهنده میزان اهمیت و تأثیر آن در کلاسبندی است. این موضوع به ما اجازه می‌دهد سوگیری‌های مدل (مثلاً نسبت به جنسیت یا نژاد) را شناسایی و رفع کنیم.
	\subsubsection*{علت}
	مفهوم 
	\lr{logistic regression}
	را در سایر درس‌ها مثل 
	\lr{ML}
	داشتیم. لذا به ذهنم آمد به جای یک سوال تکراری از شیوه کار 
	\lr{Logistic regression}
	یک سوال کمی مفهومی‌تر از مزایای آن نسبت به یک مدل پیچیده‌تر بپرسم.
